Poszukiwanie skarbów — kody

1. Wyniki

ServerScriptService → Script

Wyniki
local Gracze = game:GetService("Players")

Gracze.PlayerAdded:Connect(function(gracz)
    local wyniki = Instance.new("Folder")
    wyniki.Name = "leaderstats"
    wyniki.Parent = gracz

    local skarby = Instance.new("IntValue")
    skarby.Name = "Skarby"
    skarby.Parent = wyniki

    local wygrane = Instance.new("IntValue")
    wygrane.Name = "Wygrane"
    wygrane.Parent = wyniki
end)

2. KodSkarbu

Workspace → Skarb → Script

KodSkarbu
local Gracze = game:GetService("Players")
local skarb = script.Parent
local zebrany = false

skarb.Touched:Connect(function(dotyk)
    if zebrany then return end

    local gracz = Gracze:GetPlayerFromCharacter(dotyk.Parent)
    if not gracz then return end

    local wyniki = gracz:FindFirstChild("leaderstats")
    local licznik = wyniki and wyniki:FindFirstChild("Skarby")
    if not licznik then return end

    zebrany = true
    licznik.Value += 1
    skarb.Transparency = 1
    skarb.CanTouch = false
    skarb.CanCollide = false

    for _, obiekt in ipairs(skarb:GetDescendants()) do
        if obiekt:IsA("Decal") or obiekt:IsA("Texture") then
            obiekt.Transparency = 1
        end
    end
end)

3. KodMety

Workspace → Meta → Script

KodMety
local Gracze = game:GetService("Players")
local meta = script.Parent
local koniec = workspace:WaitForChild("Koniec")
local gotowi = {}
local cel = 5

meta.Touched:Connect(function(dotyk)
    local postac = dotyk.Parent
    local gracz = Gracze:GetPlayerFromCharacter(postac)
    if not gracz or gotowi[gracz] then return end

    local wyniki = gracz:FindFirstChild("leaderstats")
    local skarby = wyniki and wyniki:FindFirstChild("Skarby")
    local wygrane = wyniki and wyniki:FindFirstChild("Wygrane")
    if not skarby or not wygrane then return end
    if skarby.Value < cel then return end

    gotowi[gracz] = true
    wygrane.Value += 1
    postac:PivotTo(koniec.CFrame + Vector3.new(0, 4, 0))
end)

Gracze.PlayerRemoving:Connect(function(gracz)
    gotowi[gracz] = nil
end)

4. KodLawy

Workspace → Lawa → Script

KodLawy
local lawa = script.Parent

lawa.Touched:Connect(function(dotyk)
    local postac = dotyk.Parent
    local zycie = postac:FindFirstChildOfClass("Humanoid")

    if zycie then
        zycie.Health = 0
    end
end)
