Gracze.PlayerAdded:Connect(function(gracz)
    local wyniki = Instance.new("Folder")
    wyniki.Name = "leaderstats"
    wyniki.Parent = gracz

    pkt.Name = "Punkty"
    pkt.Value = 0
    pkt.Parent = wyniki

    local wygrane = Instance.new("IntValue")

    wygrane.Parent = wyniki
end)

KodMonety

Workspace → Moneta → Script

KodMonety
local Gracze = game:GetService("Players")
local moneta = script.Parent
local zebrana = false

moneta.Touched:Connect(function(dotyk)
    if zebrana then return end

    local postac = dotyk.Parent
    local gracz = Gracze:GetPlayerFromCharacter(postac)
    if not gracz then return end

    local pkt = gracz:FindFirstChild("leaderstats")
    if not pkt then return end
    pkt = pkt:FindFirstChild("Punkty")
    if not pkt then return end

    zebrana = true
    pkt.Value += 1

    local przezroczystosc = moneta.Transparency
    local naklejki = {}
    for _, obiekt in ipairs(moneta:GetDescendants()) do
        if obiekt:IsA("Decal") or obiekt:IsA("Texture") then
            naklejki[obiekt] = obiekt.Transparency
            obiekt.Transparency = 1
        end
    end

    moneta.Transparency = 1
    moneta.CanTouch = false

    task.wait(5)
    moneta.Transparency = przezroczystosc
    for obiekt, stara in pairs(naklejki) do
        if obiekt.Parent then
            obiekt.Transparency = stara
        end
    end
    moneta.CanTouch = true
    zebrana = false
end)

KodMety

Workspace → Meta → Script

KodMety
local Gracze = game:GetService("Players")
local meta = script.Parent
local konczy = {}

meta.Touched:Connect(function(dotyk)
    local gracz = Gracze:GetPlayerFromCharacter(dotyk.Parent)
    if not gracz or konczy[gracz] then return end

    local wyniki = gracz:FindFirstChild("leaderstats")
    local pkt = wyniki and wyniki:FindFirstChild("Punkty")
    local wygrane = wyniki and wyniki:FindFirstChild("Wygrane")
    if not pkt or not wygrane then return end
    if pkt.Value < 10 then return end

    konczy[gracz] = true
    pkt.Value = 0
    wygrane.Value += 1

    task.wait(2)
    if gracz.Parent then
        gracz:LoadCharacter()
    end
    konczy[gracz] = nil
end)

Gracze.PlayerRemoving:Connect(function(gracz)
    konczy[gracz] = nil
end)
